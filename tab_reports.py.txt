def safe_float(val):
    try:
        return float(str(val).replace(',', '').strip() or 0)
    except:
        return 0.0

def num_to_words(num):
    units = ["", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine", "Ten",
             "Eleven", "Twelve", "Thirteen", "Fourteen", "Fifteen", "Sixteen", "Seventeen",
             "Eighteen", "Nineteen"]
    tens = ["", "", "Twenty", "Thirty", "Forty", "Fifty", "Sixty", "Seventy", "Eighty", "Ninety"]
    if num == 0:
        return "Zero"
    def convert(n):
        if n < 20: return units[int(n)]
        if n < 100: return tens[int(n // 10)] + (" " + units[int(n % 10)] if n % 10 != 0 else "")
        if n < 1000: return units[int(n // 100)] + " Hundred" + (" And " + convert(n % 100) if n % 100 != 0 else "")
        if n < 100000: return convert(n // 1000) + " Thousand" + (" " + convert(n % 1000) if n % 1000 != 0 else "")
        if n < 10000000: return convert(n // 100000) + " Lakh" + (" " + convert(n % 100000) if n % 100000 != 0 else "")
        return convert(n // 10000000) + " Crore" + (" " + convert(n % 10000000) if n % 10000000 != 0 else "")
    return convert(abs(int(num)))

    def load_report():
        for r in tree.get_children():
            tree.delete(r)

        if p_name == "ALL":
            p_name = ""

        s_filter   = ent_status.get()

            d_from = datetime.strptime(d_from_str, "%d/%m/%Y") if d_from_str else None
        except:
            d_from = None

            d_to = datetime.strptime(d_to_str, "%d/%m/%Y") if d_to_str else None
        except:
            d_to = None

        try:
            conn = sqlite3.connect(database.DB_FILE)
            conn.row_factory = sqlite3.Row
            c = conn.cursor()

            if p_name:
                c.execute("SELECT job_no, description, type, amount FROM cost_items WHERE UPPER(description) LIKE ?",
                          (f"%{p_name}%",))
            else:
                c.execute("SELECT job_no, description, type, amount FROM cost_items")
            items = c.fetchall()

            grouped_data = {}
            for item in items:
                j_no  = item['job_no']
                party = str(item['description'] or "UNKNOWN").strip().upper()
                key   = (j_no, party)
                if key not in grouped_data:
                    grouped_data[key] = {'pay': 0.0, 'rec': 0.0}
                typ = str(item['type']).upper()
                amt = float(item['amount'] or 0)
                if typ == 'PAYABLE':
                    grouped_data[key]['pay'] += amt
                elif typ == 'RECEIVABLE':
                    grouped_data[key]['rec'] += amt

            # ADD INVOICE TOTAL AS RECEIVABLE
            try:
                c.execute("SELECT * FROM invoice_master")
                for inv in c.fetchall():
                    j_no       = str(inv['job_no'])
                    inv_no     = str(inv['invoice_no'])
                    party_name = "UNKNOWN"
                    if inv['invoice_to']:
                        party_name = str(inv['invoice_to']).split('\n')[0].strip().upper()
                    if party_name == "UNKNOWN":
                        try:
                            c.execute("SELECT consignee FROM bl_details WHERE job_no=?", (j_no,))
                            r_bl = c.fetchone()
                            if r_bl and r_bl['consignee']:
                                party_name = str(r_bl['consignee']).split('\n')[0].strip().upper()
                        except: pass
                    if p_name and p_name not in party_name:
                        continue
                    inv_amt = 0.0
                    for tbl in ['invoice_charges', 'invoice_items']:
                        try:
                            c.execute(f"SELECT total FROM {tbl} WHERE job_no=?", (j_no,))
                            inv_amt = sum(safe_float(i[0]) for i in c.fetchall())
                            if inv_amt > 0: break
                        except:
                            try:
                                c.execute(f"SELECT total FROM {tbl} WHERE invoice_no=?", (inv_no,))
                                inv_amt = sum(safe_float(i[0]) for i in c.fetchall())
                                if inv_amt > 0: break
                            except: pass
                    if inv_amt == 0:
                        try: inv_amt = safe_float(inv['grand_total'])
                        except: pass
                    if inv_amt > 0:
                        key = (j_no, party_name)
                        if key not in grouped_data:
                            grouped_data[key] = {'pay': 0.0, 'rec': 0.0}
                        grouped_data[key]['rec'] += inv_amt
            except:
                pass

            t_pay, t_rec = 0.0, 0.0

            for (j_no, party), amounts in grouped_data.items():
                payable     = round(amounts['pay'])
                receivable  = round(amounts['rec'])
                job_balance = receivable - payable

                # FIX-1: Use LIKE fuzzy match
                party_upper = party.upper()
                c.execute("""SELECT SUM(settled_amount) as s_amt FROM account_settlements
                             WHERE job_no=? AND (
                               UPPER(party_name) = ?
                               OR UPPER(?) LIKE '%' || UPPER(party_name) || '%'
                               OR UPPER(party_name) LIKE '%' || UPPER(?) || '%'
                             )""", (j_no, party_upper, party_upper, party_upper))
                s_res = c.fetchone()
                settled_so_far = round(float(s_res['s_amt'] or 0) if s_res else 0.0, 2)

                # ── STATUS LOGIC (FIXED) ──────────────────────────────────────────
                # Use round(..., 2) — keeps paisa precision, avoids integer rounding
                # errors. TOLERANCE of 0.01 ensures exact matches (e.g. 4400 == 4400)
                # never accidentally trigger EXCESS RCV due to float residues.
                abs_balance = round(abs(job_balance), 2)
                abs_settled = round(abs(settled_so_far), 2)

                # Convert to integers (cents) to completely eliminate float precision issues
                bal_cents = int(round(abs_balance * 100))
                set_cents = int(round(abs_settled * 100))

                if bal_cents == 0:
                    if set_cents > 0:
                        status = "EXCESS RCV"
                    else:
                        status = "NO DUE"
                elif set_cents > bal_cents:
                    status = "EXCESS RCV"
                elif set_cents == bal_cents:
                    status = "Settled"
                elif set_cents > 0:
                    status = "Partial Payment"
                else:
                    status = "Pending"
                # ── END STATUS LOGIC ──────────────────────────────────────────────

                if s_filter != "ALL" and status.upper() != s_filter.upper():
                    continue

                c.execute("SELECT * FROM bl_details WHERE job_no=?", (j_no,))
                bl_r = c.fetchone()
                bl = dict(bl_r) if bl_r else {}

                c.execute("SELECT * FROM cost_master WHERE job_no=?", (j_no,))
                cm_r = c.fetchone()
                cm = dict(cm_r) if cm_r else {}

                c.execute("SELECT * FROM arrival_details WHERE job_no=?", (j_no,))
                ar_r = c.fetchone()
                arr = dict(ar_r) if ar_r else {}

                j_date_str = str(bl.get('job_date', '') or cm.get('eta_date', '') or cm.get('eta', '')).strip()

                if d_from or d_to:
                    if not j_date_str:
                        continue
                    try:
                        jd = datetime.strptime(j_date_str, "%d/%m/%Y")
                        if d_from and jd < d_from:
                            continue
                        if d_to and jd > d_to:
                            continue
                    except:
                        continue

                # UPDATE-3: include in totals ONLY Pending / Partial Payment jobs
                if status.upper() in ("PENDING", "PARTIAL PAYMENT"):
                    t_pay += payable
                    t_rec += receivable

                mbl      = bl.get('mbl_no', '')
                hbl      = bl.get('hbl_no', cm.get('hbl_no', ''))
                raw_cont = bl.get('container_seal', cm.get('container_no', ''))
                if raw_cont:
                    raw_cont = str(raw_cont).strip()
                    conts = [c.strip() for c in raw_cont.replace(',', '\n').replace('&', '\n').replace('+', '\n').split('\n') if c.strip()]
                    if len(conts) > 1:
                        container_no = f"{conts[0].split('/')[0].strip()} (+)"
                    elif len(conts) == 1:
                        single = conts[0]
                        if single.count('/') > 1 or len(single) > 25:
                            container_no = f"{single.split('/')[0].strip()} (+)"
                        else:
                            container_no = single.split('/')[0].strip()
                    else:
                        container_no = ""
                else:
                    container_no = ""

                pol   = bl.get('pol', cm.get('origin', ''))
                pod   = bl.get('pod', cm.get('destination', ''))
                vsl   = arr.get('vessel', '')
                voy   = arr.get('voyage', '')

            _all_receipts.clear()

            c = conn.cursor()
            try:
                c.execute("SELECT receipt_no, date, party, total_amount FROM global_receipts")
                rows = c.fetchall()
                def parse_sort_key(row):
                    # Primary: parse receipt_no which contains YYYYMMDD-HHMMSS
                    # e.g. REC-20260503-132417 → sortable, handles same-day ordering
                    try:
                        parts = row[0].split('-')  # ['REC', '20260503', '132417']
                        if len(parts) >= 3:
                            return datetime.strptime(f"{parts[1]}-{parts[2]}", "%Y%m%d-%H%M%S")
                    except:
                        pass
                    # Fallback: parse the human-readable date string column
                    d_str = row[1]
                    try: return datetime.strptime(d_str, "%d-%b-%Y")
                    except:
                        try: return datetime.strptime(d_str, "%d/%m/%Y")
                        except: return datetime.min
                rows = sorted(rows, key=parse_sort_key)
                for r in rows:
                    rtype = "Receivable"

                        c.execute("SELECT type, amount FROM cost_items WHERE job_no=? AND UPPER(description)=?",
                                  (j, pn.upper()))
                        items = c.fetchall()
                        pay = sum(float(i[1] or 0) for i in items if str(i[0]).upper() == 'PAYABLE')
                        rec = sum(float(i[1] or 0) for i in items if str(i[0]).upper() == 'RECEIVABLE')
                        if pay > rec:
                            rtype = "Payable"
                    _all_receipts.append((r[0], r[1], r[2], f"{r[3]:,.2f}", rtype))
            except:
                pass
            conn.close()

    def download_global_report():
        if not report_data:
            messagebox.showwarning("No Data", "Please click Generate first to load data.", parent=app.root)
            return
        path = filedialog.asksaveasfilename(
            defaultextension=".xlsx",
            filetypes=[("Excel Files", "*.xlsx")],
            initialfile=f"Global_Profit_Report_{datetime.now().strftime('%d-%b-%Y')}.xlsx"
        )
        if not path:
            return
    
        # --- 1. Pre-calculate all data and summaries ---
        total_revenue = 0
        total_cost = 0
        profit_jobs = 0
        loss_jobs = 0
        processed_jobs = []
    
        for item in report_data:
            revenue = safe_float(item[9])
            cost = safe_float(item[8])
            profit = revenue - cost
    
            total_revenue += revenue
            total_cost += cost
    
            if profit > 0.01:
                profit_jobs += 1
                status = "PROFIT"
            elif profit < -0.01:
                loss_jobs += 1
                status = "LOSS"
            else:
                status = "BREAK EVEN"
    
            processed_jobs.append({
                "job_no": item[1],
                "date": item[2],
                "party": item[3],
                "hbl": item[4],
                "container": item[5],
                "pol": item[6],
                "eta": item[7],
                "revenue": revenue,
                "cost": cost,
                "profit": profit,
                "status": status
            })
    
        # --- 2. Create Excel Workbook and Styles ---
        wb = openpyxl.Workbook()
        ws = wb.active
        ws.title = "Profitability Report"
    
        # Styles
        bold_font = Font(bold=True)
        header_font = Font(bold=True, color="FFFFFF")
        header_fill = PatternFill(start_color="1E293B", end_color="1E293B", fill_type="solid")
        profit_font = Font(bold=True, color="008000")
        loss_font = Font(bold=True, color="FF0000")
        center_align = Alignment(horizontal="center")
    
        # --- 3. Write Top Summary ---
        ws['A1'] = "MK FREIGHT INTERNATIONAL"
        ws['A1'].font = Font(bold=True, size=16)
        ws['A2'] = "JOB PROFITABILITY REPORT"
        ws['A2'].font = Font(bold=True, size=13)
        ws.append([])
    
        summary_data = [
            ("Total Revenue (Receivable)", total_revenue),
            ("Total Cost (Payable)", total_cost),
            ("Gross Profit / Loss", total_revenue - total_cost),
            ("Total Jobs", len(report_data)),
            ("Profit Jobs", profit_jobs),
            ("Loss Jobs", loss_jobs)
        ]
    
        for i, (label, value) in enumerate(summary_data, 4):
            ws[f'A{i}'] = label
            ws[f'A{i}'].font = bold_font
            ws[f'B{i}'] = value
            if isinstance(value, (int, float)):
                ws[f'B{i}'].number_format = '#,##0.00'
    
        # --- 4. Write Main Table ---
        ws.append([])
        main_headers = ["JOB NO", "DATE", "PARTY", "HBL NO", "CONTAINER NO", "POL", "ETA",
                        "REVENUE", "COST", "PROFIT", "STATUS"]
        ws.append(main_headers)
        header_row_num = ws.max_row
        for cell in ws[header_row_num]:
            cell.font = header_font
            cell.fill = header_fill
            cell.alignment = center_align
    
        for job in processed_jobs:
            row_data = [
                job["job_no"], job["date"], job["party"], job["hbl"], job["container"],
                job["pol"], job["eta"], job["revenue"], job["cost"], job["profit"],
                job["status"]
            ]
            ws.append(row_data)
            current_row = ws.max_row
            for col_idx in [8, 9, 10]:
                ws.cell(row=current_row, column=col_idx).number_format = '#,##0.00'
            status_cell = ws.cell(row=current_row, column=11)
            if job["status"] == "PROFIT":
                status_cell.font = profit_font
            elif job["status"] == "LOSS":
                status_cell.font = loss_font
    
        # --- 5. Write Bottom Summary ---
        ws.append([])
        ws.append(["", "", "", "", "", "", "Total Revenue:", total_revenue])
        ws.append(["", "", "", "", "", "", "Total Cost:", total_cost])
        ws.append(["", "", "", "", "", "", "Net Profit / Loss:", total_revenue - total_cost])
    
        for i in range(3):
            row_num = ws.max_row - i
            ws.cell(row=row_num, column=7).font = bold_font
            ws.cell(row=row_num, column=8).font = bold_font
            ws.cell(row=row_num, column=8).number_format = '#,##0.00'
    
        # --- 6. Final Formatting ---
        col_widths = {'A': 15, 'B': 12, 'C': 30, 'D': 20, 'E': 20, 'F': 12, 'G': 12,
                      'H': 14, 'I': 14, 'J': 14, 'K': 12}
        for col, width in col_widths.items():
            ws.column_dimensions[col].width = width
    
        try:
            wb.save(path)
            os.startfile(path)
        except Exception as e:
            messagebox.showerror("Save Error", f"Failed to save Excel file:\n{e}", parent=app.root)
